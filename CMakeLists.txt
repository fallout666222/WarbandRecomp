cmake_minimum_required(VERSION 3.20)
project(warband_nx CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Develop on the PC first. dynarmic's A32 front end runs on both x86-64 and
# AArch64 hosts, so the same code that boots here boots on the Switch once
# the platform layer is filled in - and on the PC you have a debugger.
option(WB_SWITCH "Target Nintendo Switch (devkitA64 + libnx)" OFF)

# How much address space the guest gets, in MiB. Loading a campaign needs
# about 500 MiB of it, so 2 GiB is the working figure - but a Switch launched
# without title takeover will not give a process that much, and finding out
# which case you are in means being able to build both.
set(WB_GUEST_MB 2048 CACHE STRING "Guest address space size in MiB")

# Address sanitizer, on our own sources only - dynarmic stays uninstrumented,
# which keeps the build quick and the recompiler's generated code out of it.
#
# This is for one specific failure: a process that vanishes with no output at
# all. Windows detects heap corruption itself and calls __fastfail, which goes
# round every handler there is - no exception filter, no terminate handler,
# not even atexit. Nothing in the process can report it, so the corrupting
# write has to be caught as it happens instead.
option(WB_ASAN "Build with the address sanitizer" OFF)

# ---------------------------------------------------------------- dynarmic
# The upstream merryhime/dynarmic repository is gone. yuzu-mirror/dynarmic is
# an unofficial mirror fork of it, same 0BSD license, A32 front end intact.
set(DYNARMIC_DIR "" CACHE PATH "dynarmic source directory")
if(NOT DYNARMIC_DIR)
  foreach(candidate
      "${CMAKE_SOURCE_DIR}/../dynarmic"
      "${CMAKE_SOURCE_DIR}/../../dynarmic")
    if(EXISTS "${candidate}/CMakeLists.txt")
      get_filename_component(DYNARMIC_DIR "${candidate}" ABSOLUTE)
      break()
    endif()
  endforeach()
endif()
if(NOT DYNARMIC_DIR OR NOT EXISTS "${DYNARMIC_DIR}/CMakeLists.txt")
  message(FATAL_ERROR
    "dynarmic not found. Looked beside and one level above this project.\n"
    "  git clone https://github.com/yuzu-mirror/dynarmic\n"
    "  cmake -DDYNARMIC_DIR=/path/to/dynarmic ...\n"
    "It vendors its externals directly, so --recursive is not needed.")
endif()
message(STATUS "dynarmic: ${DYNARMIC_DIR}")

set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_FRONTENDS "A32" CACHE STRING "" FORCE)
# We build it as a subproject, so its own warnings must not fail our build.
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
if(WB_SWITCH)
  # Horizon enforces W^X for homebrew JIT: code pages are never writable and
  # executable at once. dynarmic needs to know before it emits anything.
  set(DYNARMIC_ENABLE_NO_EXECUTE_SUPPORT ON CACHE BOOL "" FORCE)
  # devkitA64's compiler is a native Windows binary driven by an MSYS2 CMake,
  # so the two disagree about what a path looks like. The toolchain file
  # translates the ones it controls; a precompiled header is generated by
  # CMake itself with an MSYS path baked in, which the compiler cannot open.
  set(DYNARMIC_USE_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
  # newlib hides fileno, fdopen and friends behind __STRICT_ANSI__, which
  # -std=c++20 turns on - and fmt, which dynarmic vendors, needs them.
  add_compile_definitions(_DEFAULT_SOURCE)
  # Cross-compiling: do not go looking for system copies of its dependencies.
  set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
  set(DYNARMIC_ENABLE_CPU_FEATURE_DETECTION OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${DYNARMIC_DIR}" dynarmic EXCLUDE_FROM_ALL)

# The vendored decoder is somebody else's code and warns like it; the rest of
# the build is -Wall -Wextra and stays that way.
if(MSVC)
  set_source_files_properties(src/third_party/stb_vorbis.c PROPERTIES
    COMPILE_OPTIONS "/W0")
else()
  set_source_files_properties(src/third_party/stb_vorbis.c PROPERTIES
    COMPILE_OPTIONS "-w")
endif()

target_include_directories(warband_nx PRIVATE src)
target_compile_definitions(warband_nx PRIVATE WB_GUEST_MB=${WB_GUEST_MB})
target_link_libraries(warband_nx PRIVATE dynarmic)
if(WIN32 AND NOT WB_SWITCH)
  # dbghelp names host frames in the crash report; without it a fault in a
  # thunk or a driver is just an address.
  # winmm is waveOut, which is the whole of the PC audio output.
  target_link_libraries(warband_nx PRIVATE opengl32 dbghelp xinput9_1_0 winmm)
endif()

if(WB_SWITCH)
  target_compile_definitions(warband_nx PRIVATE WB_SWITCH=1)

  # Horizon has no mmap and enforces W^X, so JIT pages come from libnx's Jit
  # object instead. oaknut - the assembler dynarmic's AArch64 backend emits
  # through - allocates that memory, so the Switch build puts our own copy of
  # its code_block.hpp ahead of the vendored one. BEFORE is what makes it win.
  target_include_directories(dynarmic BEFORE PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/switch/oaknut_override")

  # devkitA64 supplies libnx through its own toolchain file; point CMake at
  # $DEVKITPRO/cmake/Switch.cmake with -DCMAKE_TOOLCHAIN_FILE.
  find_library(LIBNX nx REQUIRED)

  # switch-mesa lives in portlibs, which the toolchain file makes findable but
  # does not put on the include path by itself.
  # The build runs under MSYS2's CMake, which only understands POSIX paths,
  # while DEVKITPRO is commonly set to a Windows one - and a Windows path with
  # a drive letter is silently treated as relative, producing an include
  # directory under the source tree that does not exist and an EGL/egl.h that
  # cannot be found. Translate it rather than trusting it.
  set(DKP "$ENV{DEVKITPRO}")
  if(NOT DKP)
    set(DKP "/opt/devkitpro")
  elseif(DKP MATCHES "^([A-Za-z]):[/\](.*)$")
    string(TOLOWER "${CMAKE_MATCH_1}" WB_DRIVE)
    set(DKP "/${WB_DRIVE}/${CMAKE_MATCH_2}")
  endif()
  message(STATUS "devkitPro at ${DKP}")
  target_include_directories(warband_nx PRIVATE "${DKP}/portlibs/switch/include")
  target_link_directories(warband_nx PRIVATE "${DKP}/portlibs/switch/lib")

  # Link order matters with static libraries: EGL and GLESv2 both call into
  # glapi, and glapi calls into the nouveau driver.
  target_link_libraries(warband_nx PRIVATE
    EGL GLESv2 glapi drm_nouveau ${LIBNX})

  # And the thing that actually goes on the SD card. The ELF keeps its debug
  # info - it is what symbolises a crash dump - and the .nro does not.
  nx_generate_nacp(
    OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/warband_nx.nacp"
    NAME "Mount & Blade: Warband"
    AUTHOR "warband_nx"
    VERSION "0.1.0")
  nx_create_nro(warband_nx
    NACP "${CMAKE_CURRENT_BINARY_DIR}/warband_nx.nacp")
endif()

if(MSVC)
  # Optimised, but with symbols: a release-speed run is the only one that
  # gets far enough to crash, and its stack has to be readable.
  target_compile_options(warband_nx PRIVATE /W4 /permissive- /Zi)
  target_link_options(warband_nx PRIVATE /DEBUG /OPT:REF /OPT:ICF)
  if(WB_ASAN)
    target_compile_options(warband_nx PRIVATE /fsanitize=address /Od)
    target_link_options(warband_nx PRIVATE /INCREMENTAL:NO)
    # MSVC's std::string and std::vector carry sanitizer annotations, and a
    # library built without them will not link against one built with. Only
    # our own sources are instrumented, so the annotations have to go.
    target_compile_definitions(warband_nx PRIVATE
      _DISABLE_STRING_ANNOTATION=1 _DISABLE_VECTOR_ANNOTATION=1)
  endif()
else()
  target_compile_options(warband_nx PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

# ------------------------------------------------------------- regenerate
# The thunk table is derived from the import inventory, so it can be rebuilt
# whenever the analysis changes.
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_custom_target(gen_thunks
    COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_SOURCE_DIR}/tools/gen_thunks.py"
            "${CMAKE_SOURCE_DIR}/../port_analysis/inventory.json"
            "${CMAKE_SOURCE_DIR}/src/thunks_generated.cpp"
    COMMENT "Regenerating the thunk table from inventory.json")
endif()
